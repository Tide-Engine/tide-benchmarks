cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_WIN32 AND NOT CMAKE_RC_COMPILER)
    get_filename_component(BENCH_CLANG_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
    find_program(BENCH_LLVM_RC NAMES llvm-rc HINTS "${BENCH_CLANG_BIN}")
    if(BENCH_LLVM_RC)
        set(CMAKE_RC_COMPILER "${BENCH_LLVM_RC}" CACHE FILEPATH "Resource compiler" FORCE)
    endif()
endif()
project(tide_ecs_comparison LANGUAGES C CXX)

if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "This benchmark profile requires Clang for reproducible strict floating-point flags")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(FLECS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/flecs" CACHE PATH "Pinned Flecs source directory")
set(ENTT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/entt" CACHE PATH "Pinned EnTT source directory")
if(NOT EXISTS "${FLECS_SOURCE}/distr/flecs.c" OR NOT EXISTS "${ENTT_SOURCE}/src/entt/entt.hpp")
    message(FATAL_ERROR "Pinned dependencies are missing; run scripts/build.py first")
endif()

add_library(benchmark_flags INTERFACE)
target_compile_options(benchmark_flags INTERFACE -O2 -fno-fast-math -ffp-contract=off)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|x64)$")
    target_compile_options(benchmark_flags INTERFACE -march=x86-64 -mno-fma)
endif()
target_compile_definitions(benchmark_flags INTERFACE NDEBUG)

add_library(flecs_benchmark STATIC "${FLECS_SOURCE}/distr/flecs.c")
target_include_directories(flecs_benchmark SYSTEM PUBLIC "${FLECS_SOURCE}/distr")
target_link_libraries(flecs_benchmark PRIVATE benchmark_flags)
if(WIN32)
    target_link_libraries(flecs_benchmark PUBLIC ws2_32 dbghelp)
elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(flecs_benchmark PUBLIC Threads::Threads m)
    if(NOT APPLE)
        target_compile_definitions(flecs_benchmark PRIVATE _DEFAULT_SOURCE)
    endif()
endif()

add_executable(flecs-bench cpp/flecs.cpp)
target_link_libraries(flecs-bench PRIVATE benchmark_flags flecs_benchmark)
add_executable(entt-bench cpp/entt.cpp)
target_include_directories(entt-bench SYSTEM PRIVATE "${ENTT_SOURCE}/src")
target_link_libraries(entt-bench PRIVATE benchmark_flags)
foreach(target flecs-bench entt-bench)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
